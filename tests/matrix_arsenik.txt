# Arsenik exhaustive combination matrix
# Format: test_name | +ENABLE_DEFINE -DISABLE_DEFINE ...
#   +DEFINE = uncomment (enable)
#   -DEFINE = comment out (disable)
#   omitted = keep default from options.h
#
# Defaults: ENABLE_SYMBOLS_LAYER=on, KB_LAYOUT_QWERTY=on
#           everything else off.
#
# Constraints:
#   - MAC_MODIFIERS requires ENABLE_HRM

# ── Default thumb config ──────────────────────────────────────────────

default symbols=on hrm=off              |
default symbols=off hrm=off             | -ENABLE_SYMBOLS_LAYER
default symbols=on hrm=on               | +ENABLE_HRM
default symbols=off hrm=on              | -ENABLE_SYMBOLS_LAYER +ENABLE_HRM
default symbols=on hrm=on+mac           | +ENABLE_HRM +MAC_MODIFIERS
default symbols=off hrm=on+mac          | -ENABLE_SYMBOLS_LAYER +ENABLE_HRM +MAC_MODIFIERS

# ── Vim navigation ──────────────────────────────────────────────────

vim_nav hrm=off                                        | +VIM_NAVIGATION
vim_nav symbols=off hrm=off                            | +VIM_NAVIGATION -ENABLE_SYMBOLS_LAYER
vim_nav hrm=on                                         | +VIM_NAVIGATION +ENABLE_HRM
vim_nav hrm=on+mac                                     | +VIM_NAVIGATION +ENABLE_HRM +MAC_MODIFIERS
vim_nav symbols=off hrm=on                             | +VIM_NAVIGATION -ENABLE_SYMBOLS_LAYER +ENABLE_HRM
vim_nav symbols=off hrm=on+mac                         | +VIM_NAVIGATION -ENABLE_SYMBOLS_LAYER +ENABLE_HRM +MAC_MODIFIERS

# ── Layout variations ─────────────────────────────────────────────────

layout erglace                            | -KB_LAYOUT_QWERTY +KB_LAYOUT_ERGLACE
layout bepo                               | -KB_LAYOUT_QWERTY +KB_LAYOUT_BEPO
layout dvorak                             | -KB_LAYOUT_QWERTY +KB_LAYOUT_DVORAK
layout colemak                            | -KB_LAYOUT_QWERTY +KB_LAYOUT_COLEMAK
layout workman                            | -KB_LAYOUT_QWERTY +KB_LAYOUT_WORKMAN
